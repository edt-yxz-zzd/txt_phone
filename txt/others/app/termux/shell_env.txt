#########
# 数组名 后面 必须 加 『[@]』
#########
: | : ; declare -p PIPESTATUS
  =>『declare -a PIPESTATUS=([0]="0" [1]="0")』
  定义式
: | : ; echo ${PIPESTATUS[@]@A}
  =>『declare -a PIPESTATUS=([0]="0" [1]="0")』
  同上
: | : ; echo ${PIPESTATUS[@]@Q}
  =>『'0' '0'』
  .values()
: | : ; echo "${!PIPESTATUS[@]}"
  =>『0 1』
  .keys()
: | : ; echo ${#PIPESTATUS[@]}
  =>『2』
  len()

declare -p IFS
echo ${IFS@Q}
echo ${@@Q}
printf '%s\n' "$*"
echo $?
  exit status

    太大 => n%256

  ?125?之后 shell 保留使用
  最好不要使用太大的值

  num_args#not count "$0"
