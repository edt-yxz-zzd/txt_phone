function cat5mv__myfile__conf_
{
  case 5 in
    5)
      ########ok:5
      [[ "${IFS}" == $' \t\n' ]] || { echo 1>&2 "pre-assert fail: IFS" ; exit 69 ; }
      ( IFS=$'\n' ; bash "$1" -- $(< "$2") )
          # !! (list) 拥有独立变量空间
          # vs { list ; } 共享shell变量空间
          ans=$?
      [[ "${IFS}" == $' \t\n' ]] || { echo 1>&2 "post-assert fail: IFS" ; exit 96 ; }
      return $ans
      ;;
    4)
      ########ok:4
      local -a lines
      mapfile -t lines < "$2"
      bash "$1" -- "${lines[@]}"
      return
      ;;
    3)
      ########ok:3
      local IFS=$'\n'
          bash "$1" -- $(< "$2")
      return
      ;;
    2)
      ########ok:2
      saved_IFS="${IFS}" ; IFS=$'\n' ;
          #.echo $(echo $'x y \n a ' ) ;
          #.for my_textfile_txt in $(< "$2") ; do bash "$1" -- "$my_textfile_txt" ; done
          #echo -- "$1" -- $(< "$2") ;
          bash "$1" -- $(< "$2") ;
          ans=$?
      IFS="${saved_IFS}" ;
          #.echo ${IFS@A}
      return $ans
      ;;
    1)
      ########ok:1
      xargs --arg-file="$2" -- bash "$1" --
      return
      ;;
    *)
      echo 1>&2 'bad case'
      exit 555
      ;;
      ########fail:
      #.xargs --arg-file="$2" -- "$1"
      #.  #xargs: /sdcard/0my_files//git_repos//python3_src//bash_script//app/cat5mv__myfiles: Permission denied
  esac
}
