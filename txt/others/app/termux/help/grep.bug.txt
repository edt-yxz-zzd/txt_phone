『[]』里面『\』完全无用，『]』必须在开头(除了『^』)
echo '\' | grep '[\]'
  =>『\』
<<==:
echo '[]' | grep '[\[\]]'
  =>『[]』
  ok, but:
echo '[]' | grep '[\[\] ]'
  =>『』
  <NONE>
echo '[]' | grep '[\[\][:alnum:]]'
  =>『grep: character class syntax is [[:space:]], not [:space:]』
  <ERROR>

echo '[]' | grep '[\[\]a]'
  =>『』
  <NONE>
echo '[]' | grep '[\]\[a]'
  =>『』
  <NONE>
echo '[]' | grep '[\]a\[]'
  =>『』
  <NONE>
echo '[]' | grep '[]a[]'
  =>『[]』
  ok
echo '[' | grep '[a[]'
  =>『[』
  ok
echo ']' | grep '[]a]'
  =>『]』
  ok
