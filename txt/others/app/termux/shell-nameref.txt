f() { local +x a=xxx; g a; echo "f.a: ${a@A}"; bash -c 'echo "...f.a: ${a@A}"'; declare -p a; }
g() { local +x -n b="$1"; echo "g.b: ${b@A}"; bash -c 'echo "...g.b: ${b@A}"'; declare -p b; declare -p a; b=yyy; }
unset a b; f; echo ":${a@Q},${b@Q}";
  =>:
    g.b: a='xxx'
    ...g.b:
    declare -n b="a"
    declare -- a="xxx"
    f.a: a='yyy'
    ...f.a:
    declare -- a="yyy"
    :,

f() { local -x a=xxx; g a; echo "f.a: ${a@A}"; bash -c 'echo "...f.a: ${a@A}"'; declare -p a; }
g() { local -x -n b="$1"; echo "g.b: ${b@A}"; bash -c 'echo "...g.b: ${b@A}"'; declare -p b; declare -p a; b=yyy; }
unset a b; f; echo ":${a@Q},${b@Q}";
  =>:
    g.b: declare -x a='xxx'
    ...g.b: declare -x b='a'
    declare -nx b="a"
    declare -x a="xxx"
    f.a: declare -x a='yyy'
    ...f.a: declare -x a='yyy'
    declare -x a="yyy"
    :,

局部变量牜继承属性丶值(但不包括[+-]g，强制+g):『declare -I a』
  父局部变量-->子局部变量
  全局变量-->子局部变量

父局部变量-->子局部变量
f() { local a=xxx; g a; echo "f.a: ${a@A}"; declare -p a; }
g() { declare -I a; echo "g.a: ${a@A}"; declare -p a; a=yyy; declare -p a; }
unset a b; f; echo ":${a@Q},${b@Q}";
  =>:
    g.a: a='xxx'
    declare -- a="xxx"
    declare -- a="yyy"
    f.a: a='xxx'
    declare -- a="xxx"
    :,
