我总结的case.shell_pattern语法:
[case.shell_pattern.outer ::= "(" {<pattern_list>} ")" ]
  [pattern_list ::= {<atom>}* ( "|" {<atom>}* )* ]
  [atom ::= {<plain_chars>} | {<quoted_string>} | {<escaped_char>} | {<bracket_charset>} | {<wild_chars>} | {<grouped>} ]
  # [plain_chars ::=  regex/[^$\'\"\\()?*+@!]\|[+@!]\($\|[(]\@!\)/ ]
  [plain_chars ::=  regex/[^$\'\"\\()?*]\($\|[(]\@!\)/ ]
      #排除『()』只在『case』分支模板
  [quoted_string ::=  『"..."』 | 『'...'』 ]
      #任何 展开式 包括 命令 $'...' $"..." $(...) 提前展开，不在此处
  [escaped_char ::=  regex/\\./ ]
  [bracket_charset ::=  『[...]』 ]
  [wild_chars ::=  regex/[?*]\($\|[(]\@!\)/ ]

  #only:『extglob:on』
    [grouped ::=  regex/[?*+@!]/ "(" {<pattern_list>} ")" ]
