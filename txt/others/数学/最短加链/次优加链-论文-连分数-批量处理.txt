      [g:=gcd(N,D)]
      [cf:=(N/D)的连分数展开]
      [TIME(『(N/D)的连分数展开』) == O(lnD*(1+lnN-ln(gcd(N,D))))]
      批量计算:缓存此前所有结果，所以递归计算变成查表操作{cf_chain_}
      [TIME(批量计算(cf_chain_{γ_}, [1..=N]))
      == TIME(批量计算(γ_, [1..=N]))
        +sum [TIME(『cf_chain_{递归计算变成查表操作{cf_chain_}}{γ_}(n)』) | [n:<-[1..=N]]]
      <= N*Tγ{策略;N}*TIME(__mul__{N})
        +sum [TIME(『cf_chain__{递归计算变成查表操作{cf_chain_}}{γ_}(n,d)』) | [n:<-[1..=N]][d:<-γ_(n)]]

  连分数方法
    [cf_chain__(N,D,γ_) =[def]= let [(q,r):=N/%D] in (_cf_chain__(D,r,γ_) *> cf_chain_(q,γ_) +> r)]
      前置条件:分子N{[N.bit_count()>=2][N>3]} #非二幂且非三 #即 排除 最短加链唯一 的 N
      前置条件:分母D<-[2..=N-1]
      后置条件:[返回 加链{靶D}{含D}]
        『含D』==>>『*>』的俩操作数次序
    [_cf_chain__(D,r,γ_) =[def]= if [r <- {0,1}] then cf_chain_(D,γ_) else cf_chain__(D,r,γ_)]
      前置条件:分母D<-[2..]
      前置条件:余数r<-[0..=D-1]
      后置条件:[返回 加链{靶D}][[r=!=0] -> [返回 加链{靶D}{含r}]]
      # [gcd(D,r)==gcd(N{最高层},D{最高层})]
      #     [r==0] => [D==gcd(N{最高层},D{最高层})]
      #     [r==1] => [1==gcd(N{最高层},D{最高层})]
    [cf_chain_(N,γ_) =[def]= if 欤最短加链唯一扌(N) then 唯一最短加链纟(N) else min([cf_chain__(N,D,γ_) for D in γ_(N)],key=len)]
      前置条件:N<-[1..]
    [chain8prefix *> chain7lifted =[def]= chain8prefix ++ map (chain8prefix[-1].__mul__) chain7lifted[1:]]
    [chain8prefix +> r =[def]= if r==0 then chain8prefix else assert [r <- chain8prefix] (chain8prefix ++ [chain8prefix[-1]+r])
      前置条件:r<-[0..]
      前置条件:[[r>0] -> [r<-chain8prefix]]

    [Tγ{策略;N} =[def]= max{TIME(『γ_{策略}(n)』) | [n:<-[1..=N]]} / TIME(__mul__{N})]
    ===
    [γ_{策略} :: @[N:<-{u|[u:<-[5..]][u.bit_count()>=2]}] -> 2**{2..<N}/集合纟分母]#N分子
    ===
    [Tγ{total_strategy;N} == O(N)]
    [Tγ{dyadic_strategy;N} == O(lnN)]
    [Tγ{Fermat_strategy;N} == O(lnlnN)]
    [Tγ{binary_strategy;N} == O(1)]

    ###
    [θ(N)=γ_{total_strategy}(N) =[def]= {2..<N}]
      最慢策略&&最优策略
      [11 == ℓ*(367) < ℓ{cf:θ}(367) == 12]
      [ℓ*(N) <= ℓ{cf:θ}(N) <= ℓ{cf:_}(N)]
    ###
    [β(N)=γ_{binary_strategy}(N) =[def]= {N//2}]
      批量综合表现最糟
      singleton_choice_set=>deterministic without backtrack
      二进制拆分牜甲型
      [(q,r) <- {(2,0), (2,1)}]
      模式: (D << 1) + (0|1)
      N-->D
    ###
    [α(N)=γ_{co_binary_strategy}(N) =[def]= {(N+1)//2}]
      singleton_choice_set=>deterministic without backtrack
      二进制拆分牜乙型
      [(q,r) <- {(2,0), (1,D-1)}]
      模式: (D << 1)
      模式: (D << 0) + (D-1)
        深入:模式: ((D-1)*1 +1) + (D-1)
        N-->(D-1)
    ###
    [π(N)=γ_{factor_strategy}(N) =[def]= {N-1,min_prime_factor_of(N)}\-\{N}]
      性价比最低
      不过 批量计算 时，素数筛 很快，并不会 拖后腿
    ###
    [δ(N)=γ_{dyadic_strategy}(N) =[def]= {N//2**ez | [ez:<-[1..=-1+ceil_log2(N)=-1+floor_log2(N)]]}]#注意:[N非二幂]
      性价比次高牜慢
    ###
    [F(N)=γ_{Fermat_strategy}(N) =[def]= {N//2**2**ez | [ez:<-[0..=floor_log2(-1+ceil_log2(N))=floor_log2(-1+floor_log2(N))]]}]#注意:[N非二幂]
      [[N:<-[1..=1000]] -> [ℓ{cf:F}(N) -ℓ(N) <= 1]]
      性价比最高
    ###
    [σ(N)=γ_{dichotomic_strategy}(N) =[def]= {N//2**((floor_log2(N)+1)//2)}]
      性价比次高牜快
      [σ(N)/floor_sqrt(N) ~= 1]
        # [λ(N) <= ((λ(N)+1)//2 * 2) <= 1+λ(N)]
        # [N/2 <= 2**λ(N) <= (2**((λ(N)+1)//2 * 2)) <= 2**(1+λ(N)) <= 2*N]
        # [sqrtN/sqrt2 <= (2**((λ(N)+1)//2)) <= sqrt2*sqrtN]
        # [sqrt2*sqrtN >= N/(2**((λ(N)+1)//2)) >= sqrtN/sqrt2]
        # [sqrtN/sqrt2 <= σ(N) <= sqrt2*sqrtN]
        #   [[N>=8] <-> [σ(N) >= 2]]
        #   为了避免无效d:我补丁:[[ν(N)<=3] -> [σ(N) := β(N)]]
        # [1/sqrt2 <= σ(N)/sqrtN <= sqrt2]
        #连分数拆分牜二幂冃平方根策略
        # [σ(N)~=平方根]
        #   若取:[D:=floor_sqrt(N)], 则 [q<-{D,D+1,D+2}]
        #立方根呢？其他有理数幂呢？ceil(N**(a/b))
      singleton_choice_set=>deterministic without backtrack
      the dichotomic strategy provides much shorter chains than the binary one for a large family of integers.
