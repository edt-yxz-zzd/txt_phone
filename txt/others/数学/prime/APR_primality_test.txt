Lemma 4.4.1.
  If p,q are primes with p | (q-1), then Gs(p,q)*conj_(Gs(p,q)) == q.
<==>:
[@[p,q::prime] -> [(q-1)%p == 0] -> [norm_(Gs(p,q)) == q]]
  # [norm_(x) =[def]= x*conj_(x)]

proof:
  [norm_(Gs(p,q))
  == Gs(p,q)*conj_(Gs(p,q))
  :> [X_:=DX2{p,q}]
  == sum[X_(m)*Rt{q}**m | [m:<-[1..<q]]]
   * conj_(sum[X_(m)*Rt{q}**m | [m:<-[1..<q]]])
  == sum[X_(m)*Rt{q}**m | [m:<-[1..<q]]]
   * sum[conj_(X_(m))*conj_(Rt{q})**m | [m:<-[1..<q]]]
  == sum[X_(m)*Rt{q}**m | [m:<-[1..<q]]]
   * sum[X_(m**-1%q)*(Rt{q}**-1)**m | [m:<-[1..<q]]]
  == sum[X_(m1)*X_(m2**-1%q)*Rt{q}**(m1-m2) | [m1:<-[1..<q]][m2:<-[1..<q]]]
  == sum[X_(m1*(m2**-1%q))*Rt{q}**(m1-m2) | [m1:<-[1..<q]][m2:<-[1..<q]]]
  :> [mm := (m1*(m2**-1%q))]
  => [m1 =[%q]= mm*m2]
  => [m1-m2 =[%q]= mm*m2 -m2 =[%q]= (mm-1)*m2]
  == sum[X_(mm)*Rt{q}**((mm-1)*m2) | [mm:<-[1..<q]][m2:<-[1..<q]]]
  == sum[X_(mm)*sum[Rt{q}**((mm-1)*m2) | [m2:<-[1..<q]]] | [mm:<-[1..<q]]]
  == sum[X_(mm)*sum[Rt{q}**((mm-1)*m2) | [m2:<-[1..<q]]] | [mm:<-[1..<2]]]
   + sum[X_(mm)*sum[Rt{q}**((mm-1)*m2) | [m2:<-[1..<q]]] | [mm:<-[2..<q]]]
  == X_(1)*(q-1)
   + sum[X_(mm)*sum[Rt{q}**m | [m:<-[1..<q]]] | [mm:<-[2..<q]]]
  == X_(1)*(q-1)
   + sum[X_(mm)*(-1) | [mm:<-[2..<q]]]
  == X_(1)*q
   - sum[X_(mm) | [mm:<-[1..<q]]]
  == 1*q -0
  == q
  ]
end-proof

Lemma 4.4.2.
  Suppose p,q,n are primes with p|(q-1) and gcd(p*q,n) == 1.
  Then Gs(p,q)**(n**(p-1)-1) ≡ χ{p,q}(n) (mod n).
<==>:
[@[n,p,q::prime] -> [(q-1)%p == 0] -> [gcd(p*q,n) == 1] -> [Gs(p,q)**(n**(p-1)-1) =[%n]= DX2{p,q}(n)]]
proof:
  [Gs(p,q)**(n**(p-1))
  :> [X_:=DX2{p,q}]
  == sum[X_(m)*Rt{q}**m | [m:<-[1..<q]]]**(n**(p-1))
  !! [n::prime]
  =[%n]= sum[X_(m)**(n**(p-1)) * (Rt{q}**m)**(n**(p-1)) | [m:<-[1..<q]]]
  !! [order_of(DX2{p,q}) == p]
  == sum[X_(m)**(n**(p-1) %p) * Rt{q}**(m*(n**(p-1))) | [m:<-[1..<q]]]
  !! [p::prime]
  !! [n%p =!= 0]
  => [n**(p-1) =[%p]= 1]
  == sum[X_(m) * Rt{q}**(m*(n**(p-1))) | [m:<-[1..<q]]]
  == sum[X_(m*(n**(p-1)) * (n**-1%q)**(p-1)) * Rt{q}**(m*(n**(p-1))) | [m:<-[1..<q]]]
  == sum[X_(m*(n**(p-1))) * X_((n**-1%q)**(p-1)) * Rt{q}**(m*(n**(p-1))) | [m:<-[1..<q]]]
  == X_((n**-1%q)**(p-1)) * sum[X_(m*(n**(p-1))) * Rt{q}**(m*(n**(p-1))) | [m:<-[1..<q]]]
  !! [n%q =!= 0]
  => [n**(p-1)%q =!= 0]
  == X_((n**-1%q)**(p-1)) * sum[X_(j) * Rt{q}**j | [j:<-[1..<q]]]
  == X_((n**-1%q)**(p-1)) * Gs(p,q)
  == X_((n**-1%q)**p * n) * Gs(p,q)
  == X_(n**-1%q)**p * X_(n) * Gs(p,q)
  !! [order_of(DX2{p,q}) == p]
  == 1 * X_(n) * Gs(p,q)
  == X_(n) * Gs(p,q)
  ]
  [Gs(p,q)**(n**(p-1)) =[%n]= DX2{p,q}(n) * Gs(p,q)]
  [Gs(p,q)**(n**(p-1))*conj_(Gs(p,q)) =[%n]= DX2{p,q}(n) * Gs(p,q)*conj_(Gs(p,q))]
  [Gs(p,q)**(n**(p-1)-1)*norm_(Gs(p,q)) =[%n]= DX2{p,q}(n) *norm_(Gs(p,q))]
  !! [@[p,q::prime] -> [(q-1)%p == 0] -> [norm_(Gs(p,q)) == q]]
  [Gs(p,q)**(n**(p-1)-1)*q =[%n]= DX2{p,q}(n) *q]
  !! [gcd(q,n) == 1]
  [Gs(p,q)**(n**(p-1)-1) =[%n]= DX2{p,q}(n)]
end-proof

The final method is presently the only one that leads to rigorous primality proofs.
  It consists of subjecting n to a series of tests, similar to those in (1.2) and (1.3), with the following two properties.
    First, if n is prime then it passes the tests.
    Secondly, if n passes the tests, then information is obtained about the possible divisors of n.
      This information should eventually lead to the conclusion that 1 and n are the only divisors of n, so that n is prime.

To describe the type of information that is obtained, we let
  H be a group, and
  Ψ a map from the set of divisors of n to H
    with the property that
      Ψ(r*r') = Ψ(r)*Ψ(r') if r*r' divides n.
  If n passes the tests, then it follows that for suitable choices of H and Ψ we have
    (1.4) Ψ(r) is a power of Ψ(n), for every divisor r of n.
