黄金比例
[φ == phi == golden_ratio == (-1+sqrt5)/2 ~= 0.6180339887498949]
[1 > phi > (1-phi)]
[1/2 < phi < 1]
[phi/1 == (1-phi)/phi == 1/phi -1]
[1+phi == 1/phi == phi/(1-phi)]
[phi**2 == (1-phi)]
[phi**2 +phi -1 == 0]
[phi == (-1+[+-]sqrt5)/2]
!! [1/2 < phi < 1]
[phi == (-1+sqrt5)/2]
[1/phi == (1+sqrt5)/2 == 1+phi]
[1/φ == inv_phi == inv_golden_ratio == (1+sqrt5)/2 ~= 1.618033988749895]

[fibonacci[n] == 1/sqrt5*(inv_golden_ratio**n -(-golden_ratio)**n)]
[fibonacci[n] == (inv_golden_ratio**n -neg_golden_ratio**n) / (inv_golden_ratio**1-neg_golden_ratio**1)]

[[x > 1] -> [floor_log_(47;x**8) < log_(inv_golden_ratio; x) < ceil_log_(76;x**9)]]

from math import log
from seed.math.floor_ceil_log__via_div_log2 import FloorLogarithm, floor_log__via_div_log2_, floor_ceil_log__via_div_log2_, ceil_log__via_div_log2_
inv_golden_ratio = 1.618033988749895
log_ = lambda b, x, /:log(x, b)
logIG = lambda x, /:log(x, inv_golden_ratio)
_103680 = FloorLogarithm(103680)
_271445 = FloorLogarithm(271445)
def test_(x, /):
    assert x > 1
    r = log(x, inv_golden_ratio)
    # !! [[x > 1] -> [floor_log_(271445;x**26) < log_(inv_golden_ratio; x) < ceil_log_(103680;x**24)]]
    x2 = x**2
    x24 = x2**12
    x26 = x2*x24
    assert (LB:=_271445.floor_log_(x26, with_floor_pow=False)) < r, (x, r, LB)
    assert (UB:=_103680.ceil_log_(x24, with_floor_pow=False)) > r, (x, r, LB, UB)

for x in range(2, 3+2**17):test_(x)
  ok!
